## 一、痛点分析：我为什么放弃云服务？

### ❌ 使用云端服务的烦恼

| 问题 | OpenAI + Pinecone | 国内云服务 |
|------|------------------|-----------|
| 💰 费用 | 按 token 计费，贵！ | 按调用次数收费 |
| 🔒 隐私 | 数据上传到第三方 | 数据需出境 |
| 🌐 网络 | 需科学上网 | 国内访问尚可 |
| ⏱️ 延迟 | 200-500ms | 100-300ms |
| 📦 依赖 | 太多外部依赖 | 服务不稳定 |

### ✅ 本地部署的优势

```
完全免费 + 数据安全 + 无需联网 + 响应更快
```

**部署成本：**
- 💻 一台普通电脑（8GB+ 内存）
- 🐳 Docker Desktop（免费）

```go
// pkg/embedding/client.go
package embedding

import (
    "bytes"
    "context"
    "encoding/json"
    "fmt"
    "io"
    "net/http"
)

type Client struct{ Model string }

func NewClient(model string) *Client {
    return &Client{Model: model}
}

// Embedding 单条文本向量化
func (c *Client) Embedding(ctx context.Context, text string) ([]float32, error) {
    reqBody := map[string]interface{}{"model": c.Model, "input": text}
    reqBytes, _ := json.Marshal(reqBody)

    req, _ := http.NewRequestWithContext(ctx, "POST",
        "http://localhost:11434/api/embed", bytes.NewReader(reqBytes))
    req.Header.Set("Content-Type", "application/json")

    resp, err := http.DefaultClient.Do(req)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()

    var result struct {
        Embeddings [][]float32 `json:"embeddings"`
    }
    json.NewDecoder(resp.Body).Decode(&result)

    if len(result.Embeddings) > 0 {
        return result.Embeddings[0], nil
    }
    return nil, fmt.Errorf("no embedding returned")
}
```

**Milvus 客户端：**

```go
// pkg/vector/client.go
package vector

import (
    "context"
    "fmt"

    "github.com/milvus-io/milvus-sdk-go/v2/client"
    "github.com/milvus-io/milvus-sdk-go/v2/entity"
)

type Config struct {
    Addr       string
    Collection string
    Dim        int
}

type Client struct {
    cfg    Config
    milvus client.Client
}

// 创建向量客户端
func NewClient(cfg Config) (*Client, error) {
    c, err := client.NewClient(context.Background(), client.Config{Address: cfg.Addr})
    if err != nil {
        return nil, err
    }
    return &Client{cfg: cfg, milvus: c}, nil
}

// 创建集合
func (c *Client) CreateCollection(ctx context.Context) error {
    schema := entity.NewSchema().WithName(c.cfg.Collection).
        WithField(entity.NewField().WithName("id").WithDataType(entity.FieldTypeInt64).WithIsPrimaryKey(true)).
        WithField(entity.NewField().WithName("vector").WithDataType(entity.FieldTypeFloatVector).WithDim(int64(c.cfg.Dim))).
        WithField(entity.NewField().WithName("text").WithDataType(entity.FieldTypeVarChar).WithMaxLength(65535))

    c.milvus.CreateCollection(ctx, schema, 1)

    // 创建索引
    idx, _ := entity.NewIndexHNSW(entity.L2, 16, 128)
    c.milvus.CreateIndex(ctx, c.cfg.Collection, "vector", idx, false)

    // 加载到内存
    c.milvus.LoadCollection(ctx, c.cfg.Collection, false)
    return nil
}

// 插入数据
func (c *Client) Insert(ctx context.Context, id int64, vec []float32, text string) error {
    colID := entity.NewColumnInt64("id", []int64{id})
    colVec := entity.NewColumnFloatVector("vector", c.cfg.Dim, [][]float32{vec})
    colText := entity.NewColumnVarChar("text", []string{text})

    _, err := c.milvus.Insert(ctx, c.cfg.Collection, "", colID, colVec, colText)
    c.milvus.Flush(ctx, c.cfg.Collection, true)
    return err
}

// 搜索
func (c *Client) Search(ctx context.Context, query []float32, limit int) ([]float32, []string, error) {
    sp, _ := entity.NewIndexHNSWSearchParam(128)
    results, err := c.milvus.Search(ctx, c.cfg.Collection, nil, "",
        []string{"id", "text"}, []entity.Vector{entity.FloatVector(query)},
        "vector", entity.L2, int64(limit), sp)

    if err != nil {
        return nil, nil, err
    }

    scores := results[0].Scores
    texts := make([]string, 0, len(results[0].Fields))
    for _, f := range results[0].Fields {
        if f.Name() == "text" {
            col := f.(*entity.ColumnVarChar)
            for i := 0; i < col.Len(); i++ {
                texts = append(texts, col.Value().([]string)[i])
            }
        }
    }
    return scores, texts, nil
}

func (c *Client) Close() error {
    return c.milvus.Close()
}
```

**命令行工具：**

```go
// cmd/vector/main.go
package main

import (
    "context"
    "fmt"
    "log"
    "os"
    "strconv"

    "golllm/pkg/embedding"
    "golllm/pkg/vector"
)

func main() {
    ctx := context.Background()

    // 初始化客户端
    vc, _ := vector.NewClient(vector.Config{
        Addr:       "localhost:19530",
        Collection: "documents",
        Dim:        768,
    })
    emb := embedding.NewClient("nomic-embed-text")
    defer vc.Close()

    // 命令处理
    switch os.Args[1] {
    case "create":
        vc.CreateCollection(ctx)
        fmt.Println("✅ 集合创建成功！")

    case "insert":
        id, _ := strconv.ParseInt(os.Args[2], 10, 64)
        text := os.Args[3]
        vec, _ := emb.Embedding(ctx, text)
        vc.Insert(ctx, id, vec, text)
        fmt.Printf("✅ 插入成功！ID=%d\n", id)

    case "search":
        query := os.Args[2]
        limit, _ := strconv.Atoi(os.Args[3])
        vec, _ := emb.Embedding(ctx, query)
        scores, texts, _ := vc.Search(ctx, vec, limit)

        fmt.Printf("\n🔍 搜索「%s」，找到 %d 条结果：\n\n", query, len(texts))
        for i := 0; i < len(scores); i++ {
            fmt.Printf("%d️⃣ Score: %.4f\n   %s\n\n", i+1, scores[i], texts[i])
        }
    }
}
```

## 七、常见问题 Q&A

### Q1：Milvus 启动失败，提示端口被占用？

```powershell

ports:
  - "29530:19530"  # 外部 29530 映射到容器内 19530
```


### Q4：向量维度不匹配？

```bash
# 原因：Embedding 模型维度 ≠ Milvus 集合维度
# nomic-embed-text: 768 维
# bge-m3: 1024 维

# 解决方案：创建集合时指定正确的 Dim
vc := NewClient(Config{Dim: 768})  # nomic-embed-text
vc := NewClient(Config{Dim: 1024}) # bge-m3
```

```bash
# 可能原因：

# 2. 数据未 flush → 插入后调用 Flush
# 3. 索引未创建 → 必须先创建索引才能搜索

# 强制加载集合
docker compose exec standalone milvusctl load collection
```
